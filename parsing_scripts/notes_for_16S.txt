#!/usr/bin/env python3
import gzip
import sys

def parse_fastq_gz(filename):
    """Generator to parse gzipped FASTQ"""
    with gzip.open(filename, 'rt') as f:
        while True:
            header = f.readline().strip()
            if not header:
                break
            seq = f.readline().strip()
            plus = f.readline().strip()
            qual = f.readline().strip()
            yield (header, seq, plus, qual)

def write_fastq_record(f, header, seq, plus, qual):
    """Write a FASTQ record"""
    f.write(f"{header}\n{seq}\n{plus}\n{qual}\n")

# Load captured barcode+UMI pairs
print("Loading captured barcodes and UMIs...", file=sys.stderr)
captured = set()
with open(sys.argv[1]) as f:  # captured_reads.txt
    for line in f:
        parts = line.strip().split('\t')
        barcode = parts[0]
        umi = parts[1]
        captured.add((barcode, umi))

print(f"Loaded {len(captured)} unique barcode-UMI combinations", file=sys.stderr)

# Filter FASTQs
r1_in = sys.argv[2]  # R1.fastq.gz
r2_in = sys.argv[3]  # R2.fastq.gz
r1_out = sys.argv[4]  # filtered_R1.fastq.gz
r2_out = sys.argv[5]  # filtered_R2.fastq.gz

matched = 0
total = 0

with gzip.open(r1_out, 'wt') as out1, gzip.open(r2_out, 'wt') as out2:
    for (h1, s1, p1, q1), (h2, s2, p2, q2) in zip(
        parse_fastq_gz(r1_in), 
        parse_fastq_gz(r2_in)
    ):
        total += 1
        
        # Extract barcode and UMI from R1
        barcode = s1[0:16]
        umi = s1[16:28]
        
        if (barcode, umi) in captured:
            write_fastq_record(out1, h1, s1, p1, q1)
            write_fastq_record(out2, h2, s2, p2, q2)
            matched += 1
        
        if total % 1000000 == 0:
            print(f"Processed {total:,} reads, matched {matched:,}", file=sys.stderr)

print(f"\nDone! Matched {matched:,} / {total:,} reads ({100*matched/total:.2f}%)", file=sys.stderr)
